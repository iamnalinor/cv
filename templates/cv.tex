% Custom commands
\newcommand{\resumeItem}[1]{ \item\small{ { #1 \vspace{-2pt}} } }

\newcommand{\resumeExperienceItem}[5]{
\vspace{-2pt}
\item
\begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
  \textbf{ #1} $|$ \emph{ #2} & #3                 \\
  \textit{\small#4}           & \textit{\small #5} \\
\end{tabular*}
\vspace{-7pt}
}

\newcommand{\resumeEducationItem}[4]{
\vspace{-2pt}
\item
\begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
  \textbf{ #1}      & #2                 \\
  \textit{\small#3} & \textit{\small #4} \\
\end{tabular*}
\vspace{-7pt}
}

\newcommand{\resumeShortExperienceItem}[3]{ \item
\begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
  \textbf{ #1} $|$ \emph{ #2} & #3 \\
\end{tabular*}
\vspace{-7pt}
}

\newcommand{\resumeExperienceItemListEnd}{\end{itemize}}

\newcommand{\resumeItemListEnd}{\end{itemize}

    \textbf{\Huge \scshape {{ header.first_name }} {{ header.last_name }}} \\
    \vspace{1pt} \small
    {% if header.contacts %}
      \begin{tabular}{c}
        {% for type, contact in header.contacts.items() %}
          {% if type == "github" %}
            \raisebox{-0.05\height}{\faGithub} \href{https://github.com/{{ contact }}}{ {{ contact }} }
          {% elif type == "telegram" %}
            \raisebox{-0.05\height}{\faTelegram} \href{https://t.me/{{ contact }}}{ @{{ contact }} }
          {% elif type == "email" %}
            \raisebox{-0.05\height}{\faEnvelope} \href{mailto:{{ contact }}}{ {{ contact }} }
          {% endif %}
        {% endfor %}
      \end{tabular}
    {% endif %}
     
  \end{center}

  {% for block in blocks %}

    {% if not block.skip_header %}
    \section{ {{ block.name }} }
    {% endif %}

    {% if block.type == "text" %}

      \small {{ block.content }}

    {% elif block.type == "list" %}

      \begin{itemize}[noitemsep]
        {% for point in block.content %}
          \item {{ point }}
        {% endfor %}
      \end{itemize}

    {% elif block.type == "mapping" %}

      \begin{itemize}[leftmargin=0.15in, label={}]
        \small{\item{
          {% for item in block.content %}\textbf{\phantom{}{{ item.name }}}{: {{ item.value }}} \\ {% endfor %}
        }}
      \end{itemize}

    {% elif block.type == "education" %}
      \resumeExperienceItemListStart

      {% for item in block.content %}
        \resumeEducationItem{\phantom{}{{ item.name }}}{\phantom{}{{ item.location }}}{\phantom{}{{ item.degree }}}{\phantom{}{{ item.period }}}
      {% endfor %}

      \resumeExperienceItemListEnd

    {% elif block.type == "experience" %}
      \resumeExperienceItemListStart

      {% for item in block.content %}
        {% if item.company and item.location %}\resumeExperienceItem{% else %}\resumeShortExperienceItem{% endif %}
        {\phantom{}{{ item.name }}}{\phantom{}{{ item.stack }}}{\phantom{}{{ item.period }}}
        % Those two spaces before item.company are important
        {% if item.company and item.location %}{  {{ item.company }}}{\phantom{}{{ item.location }}}{% endif %}

        {% if item.points %}
          \resumeItemListStart
            {% for point in item.points %}
              \resumeItem{ {{ point }} }
            {% endfor %}
          \resumeItemListEnd
        {% endif %}
      {% endfor %}

      \resumeExperienceItemListEnd
    {% endif %}
  {% endfor %}

\end{document}
